  const s = await window.qwen.invoke('server:start');

  const st = await window.qwen.invoke('server:status');

  document.getElementById('prompt').value = 'um cartoon colorido de uma raposa com lenco vermelho, estilo aquarela';
  document.getElementById('steps').value = '6';

  });
