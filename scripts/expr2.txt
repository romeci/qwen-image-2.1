(async () => {
  const out = {};
  // 1) clica na thumb: deve SO trocar a foto do painel, modal continua fechado
  const thumb = document.querySelectorAll('#gallery img')[1] || document.querySelectorAll('#gallery img')[0];
  thumb.click();
  out.preview = { srcOk: document.getElementById('result').src.startsWith('file:'),
    resultWrapVisible: !document.getElementById('resultWrap').classList.contains('hidden'),
    modalFechado: document.getElementById('modal').classList.contains('hidden') };
  // 2) galeria vertical
  const g = document.getElementById('gallery');
  const cs = getComputedStyle(g);
  out.galeria = { cols: cs.gridTemplateColumns.split(' ').length,
    overflowX: cs.overflowX, overflowY: cs.overflowY };
  // 3) clica "Editar": modal abre SO em modo edicao (com os chips de marca)
  document.getElementById('btnEditResult').click();
  await new Promise(r => setTimeout(r, 700));
  out.edit = { modalAberto: !document.getElementById('modal').classList.contains('hidden'),
    editVis: !document.getElementById('modalEdit').classList.contains('hidden'),
    viewEscondido: document.getElementById('modalView').classList.contains('hidden'),
    marks: !!document.getElementById('markEdit') };
  return JSON.stringify(out);
})()
