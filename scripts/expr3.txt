(async () => {
  const out = {};
  out.scroll = { body: document.body.scrollHeight <= window.innerHeight + 1,
    doc: document.documentElement.scrollHeight <= window.innerHeight + 1,
    vh: window.innerHeight, bodyH: document.body.scrollHeight };
  let comBarra = 0;
  document.querySelectorAll('*').forEach((el) => {
    const cs = getComputedStyle(el);
    if ((el.scrollHeight > el.clientHeight + 2 && ['auto','scroll'].includes(cs.overflowY))
      || (el.scrollWidth > el.clientWidth + 2 && ['auto','scroll'].includes(cs.overflowX))) {
      if (el.offsetWidth - el.clientWidth > 2) comBarra++;
    }
  });
  out.scroll.barrasVisiveis = comBarra;
  const g = document.getElementById('gallery');
  const cs = getComputedStyle(g);
  out.galeria = { cols: cs.gridTemplateColumns.split(' ').length, overflowX: cs.overflowX };
  const items = window.__gallery || [];
  out.ordem = { total: items.length,
    desc: items.every((it, i) => i === 0 || items[i - 1].mtime >= it.mtime) };
  const imgs = document.querySelectorAll('#gallery img');
  imgs[0].click(); imgs[1] && imgs[1].click();
  out.sel2 = { n: window.__state.sel.size,
    editar: !document.getElementById('btnSelEdit').disabled,
    explorer: !document.getElementById('btnSelExplorer').disabled,
    excluir: !document.getElementById('btnSelDelete').disabled,
    badge: document.getElementById('galleryCount').textContent };
  document.getElementById('btnSelectAll').click();
  out.selAll = { n: window.__state.sel.size, total: items.length,
    editar: !document.getElementById('btnSelEdit').disabled,
    explorer: !document.getElementById('btnSelExplorer').disabled,
    excluir: !document.getElementById('btnSelDelete').disabled };
  document.getElementById('btnSelectAll').click();
  out.refsUI = { count: document.getElementById('mainRefsCount').textContent };
  return JSON.stringify(out);
})()
