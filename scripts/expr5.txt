(async () => {
  const out = {};
  await window.__refresh();
  // ---------- 1) EXCLUSAO REAL ----------
  const imgs = [...document.querySelectorAll('#gallery img')];
  const lixo = imgs.find((i) => i.title === 'lixo_teste_zz.png');
  if (!lixo) return JSON.stringify({ erro: 'lixo_teste_zz.png nao apareceu', titulos: imgs.map((i) => i.title).slice(0, 6) });
  lixo.click();
  out.delete = { selecionado: window.__state.sel.size, excluirHabilitado: !document.getElementById('btnSelDelete').disabled };
  document.getElementById('btnSelDelete').click();
  await new Promise((r) => setTimeout(r, 4000));
  await window.__refresh();
  out.delete.aindaTem = (window.__gallery || []).some((i) => i.filename === 'lixo_teste_zz.png');
  out.delete.selApos = window.__state.sel.size;
  out.delete.log = document.getElementById('log').textContent.split('\n').filter((l) => l.includes('[del]') || l.includes('lixeira')).slice(-2).join(' | ');

  // ---------- 2) REFS NO t2I (2 refs) ----------
  const g = window.__gallery || [];
  const refs = [g.find((i) => i.filename.startsWith('Qwen_t2i'))?.path, g.find((i) => i.filename.startsWith('Qwen_edit'))?.path].filter(Boolean);
  window.__state.refs = refs;
  window.__renderMainRefs();
  out.refs = { n: refs.length, count: document.getElementById('mainRefsCount').textContent };

  // ---------- 3) gerar SEM prompt (so refs) ----------
  window.__lastGenResult = null;
  document.getElementById('prompt').value = '';
  document.getElementById('steps').value = '6';
  document.getElementById('error').classList.add('hidden');
  window.__genT2i();
  const r = await new Promise((res) => {
    const iv = setInterval(() => { if (window.__lastGenResult) { clearInterval(iv); res(window.__lastGenResult); } }, 2000);
    setTimeout(() => { clearInterval(iv); res({ timeout: true }); }, 90000);
  });
  out.genSemPrompt = { ok: r.ok, files: (r.files || []).map((f) => f.filename), error: r.error || null,
    erroUI: document.getElementById('error').textContent.slice(0, 80) };
  return JSON.stringify(out);
})()
