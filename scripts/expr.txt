(async () => {
  window.__lastGenResult = null;
  document.getElementById('prompt').value = 'teste de cancelamento: paisagem de montanha ao amanhecer';
  document.getElementById('steps').value = '40';
  const t0 = Date.now();
  window.__genT2i();
  await new Promise(r => setTimeout(r, 10000));
  const tc = Date.now();
  await window.qwen.invoke('gen:cancel');
  const r = await new Promise((res) => {
    const iv = setInterval(() => { if (window.__lastGenResult) { clearInterval(iv); res(window.__lastGenResult); } }, 400);

  });
  return JSON.stringify({ cancelRespMs: Date.now() - tc, r,
    err: document.getElementById('error').textContent.slice(0, 90),
    genBtnHabilitado: !document.getElementById('btnGen').disabled,
    logFim: document.getElementById('log').textContent.slice(-260) });
